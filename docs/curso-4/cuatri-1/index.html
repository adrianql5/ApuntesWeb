-- ===================================================================== -- Almacén de datos: rendimiento deportivo en el circuito IJF -- Modelo dimensional en estrella (PostgreSQL) -- Granularidad: una fila por participación de un judoka en un -- campeonato en una fecha. -- =====================================================================

DROP TABLE IF EXISTS hechos_participacion; DROP TABLE IF EXISTS dim_judoka; DROP TABLE IF EXISTS dim_campeonato; DROP TABLE IF EXISTS dim_categoria_peso; DROP TABLE IF EXISTS dim_categoria_edad; DROP TABLE IF EXISTS dim_fecha;

-- DIMENSIÓN FECHA -- Jerarquía: día -> mes -> trimestre -> año -> ciclo olímpico

CREATE TABLE dim_fecha ( fecha_id INTEGER PRIMARY KEY, -- formato AAAAMMDD fecha DATE NOT NULL UNIQUE, dia SMALLINT NOT NULL CHECK (dia BETWEEN 1 AND 31), dia_semana VARCHAR(10) NOT NULL, -- 'Lunes', ... mes SMALLINT NOT NULL CHECK (mes BETWEEN 1 AND 12), nombre_mes VARCHAR(12) NOT NULL, -- 'Enero', ... trimestre SMALLINT NOT NULL CHECK (trimestre BETWEEN 1 AND 4), anio SMALLINT NOT NULL, ciclo_olimpico VARCHAR(9) NOT NULL -- '2025-2028' );

-- DIMENSIÓN CATEGORÍA DE EDAD NATURAL DEL JUDOKA -- Categoría que le corresponde por edad el día de la competición. -- Sin jerarquía; 'orden' permite compararla con la del campeonato.

CREATE TABLE dim_categoria_edad ( categoria_edad_id SMALLINT PRIMARY KEY, nombre VARCHAR(10) NOT NULL UNIQUE, -- 'Cadete', 'Júnior', 'Sénior' orden SMALLINT NOT NULL UNIQUE CHECK (orden BETWEEN 1 AND 3), edad_natural_min SMALLINT NOT NULL, edad_natural_max SMALLINT -- NULL = sin límite (sénior) );

-- DIMENSIÓN CATEGORÍA DE PESO -- Jerarquía: categoría -> grupo de peso -> género

CREATE TABLE dim_categoria_peso ( categoria_peso_id SMALLINT PRIMARY KEY, nombre VARCHAR(10) NOT NULL, -- '-73 kg', '+100 kg' genero CHAR(1) NOT NULL CHECK (genero IN ('M', 'F')), limite_kg NUMERIC(5,1), -- NULL para '+' (sin límite) grupo_peso VARCHAR(10) NOT NULL CHECK (grupo_peso IN ('Ligero', 'Medio', 'Pesado')), UNIQUE (nombre, genero) );

-- DIMENSIÓN CAMPEONATO (una fila por campeonato; la edición concreta -- queda identificada por campeonato + fecha en la tabla de hechos) -- Jerarquías: ciudad -> país -> continente -- campeonato -> rango

CREATE TABLE dim_campeonato ( campeonato_id SERIAL PRIMARY KEY, -- clave subrogada nombre_oficial VARCHAR(100) NOT NULL UNIQUE, -- 'Grand Slam de París' rango VARCHAR(30) NOT NULL, -- 'Grand Slam', 'Grand Prix', 'Masters', -- 'Campeonato del Mundo', 'Continental Open'... categoria_edad VARCHAR(10) NOT NULL CHECK (categoria_edad IN ('Cadete', 'Júnior', 'Sénior')), categoria_edad_orden SMALLINT NOT NULL CHECK (categoria_edad_orden BETWEEN 1 AND 3), ciudad VARCHAR(60) NOT NULL, pais VARCHAR(60) NOT NULL, continente VARCHAR(20) NOT NULL );

-- DIMENSIÓN JUDOKA -- Atributos estables: SCD tipo 0. -- Cinturón: SCD tipo 2 (nueva fila con nueva clave subrogada). -- Jerarquía: judoka -> federación -> unión continental

CREATE TABLE dim_judoka ( judoka_sk SERIAL PRIMARY KEY, -- clave subrogada judoka_id_ijf INTEGER NOT NULL, -- clave natural (sistema IJF) nombre_completo VARCHAR(120) NOT NULL, genero CHAR(1) NOT NULL CHECK (genero IN ('M', 'F')), fecha_nacimiento DATE NOT NULL, federacion_codigo CHAR(3) NOT NULL, -- 'ESP', 'JPN', ... federacion_nombre VARCHAR(100) NOT NULL, union_continental VARCHAR(5) NOT NULL, -- 'EJU', 'JUA', 'PJC', 'AJU', 'OJU' cinturon VARCHAR(15) NOT NULL, -- '1º Dan', '2º Dan', ... fecha_inicio DATE NOT NULL, fecha_fin DATE NOT NULL DEFAULT DATE '9999-12-31', es_actual BOOLEAN NOT NULL DEFAULT TRUE, UNIQUE (judoka_id_ijf, fecha_inicio), CHECK (fecha_fin >= fecha_inicio) );

-- TABLA DE HECHOS: PARTICIPACIÓN

CREATE TABLE hechos_participacion ( -- Claves foráneas a dimensiones fecha_id INTEGER NOT NULL REFERENCES dim_fecha (fecha_id), campeonato_id INTEGER NOT NULL REFERENCES dim_campeonato (campeonato_id), judoka_sk INTEGER NOT NULL REFERENCES dim_judoka (judoka_sk), categoria_peso_id SMALLINT NOT NULL REFERENCES dim_categoria_peso (categoria_peso_id), categoria_edad_judoka_id SMALLINT NOT NULL REFERENCES dim_categoria_edad (categoria_edad_id),

-- Medidas aditivas
puntos               INTEGER   NOT NULL DEFAULT 0 CHECK (puntos >= 0),
combates_disputados  SMALLINT  NOT NULL CHECK (combates_disputados >= 0),
combates_ganados     SMALLINT  NOT NULL CHECK (combates_ganados >= 0),
es_oro               SMALLINT  NOT NULL DEFAULT 0 CHECK (es_oro IN (0, 1)),
es_plata             SMALLINT  NOT NULL DEFAULT 0 CHECK (es_plata IN (0, 1)),
es_bronce            SMALLINT  NOT NULL DEFAULT 0 CHECK (es_bronce IN (0, 1)),
fuera_categoria      SMALLINT  NOT NULL DEFAULT 0 CHECK (fuera_categoria IN (0, 1)),

-- Medidas no aditivas
posicion             SMALLINT  NOT NULL CHECK (posicion >= 1),
edad                 SMALLINT  NOT NULL CHECK (edad > 0),

-- Granularidad: un judoka, en un campeonato, en una fecha
PRIMARY KEY (campeonato_id, fecha_id, judoka_sk),
CHECK (combates_ganados <= combates_disputados),
CHECK (es_oro + es_plata + es_bronce <= 1)

);

-- Índices sobre las claves foráneas para acelerar los JOIN CREATE INDEX idx_hp_fecha ON hechos_participacion (fecha_id); CREATE INDEX idx_hp_judoka ON hechos_participacion (judoka_sk); CREATE INDEX idx_hp_peso ON hechos_participacion (categoria_peso_id); CREATE INDEX idx_hp_edad_jud ON hechos_participacion (categoria_edad_judoka_id);
